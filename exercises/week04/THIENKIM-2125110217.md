CÂU TRẢ LỜI:
# ==========================
# BUBBLE SORT
# ==========================
def bubble_sort(arr):

    print("\n===== BUBBLE SORT =====")
    print("Ban đầu:", arr)

    step = 1
    n = len(arr)

    for i in range(n):

        for j in range(n - i - 1):

            if arr[j] > arr[j + 1]:

                arr[j], arr[j + 1] = arr[j + 1], arr[j]

                print(
                    f"Bước {step}: {arr}"
                )

                step += 1

    print("Kết quả:", arr)


# ==========================
# SELECTION SORT
# ==========================
def selection_sort(arr):

    print("\n===== SELECTION SORT =====")
    print("Ban đầu:", arr)

    step = 1
    n = len(arr)

    for i in range(n):

        min_idx = i

        for j in range(i + 1, n):

            if arr[j] < arr[min_idx]:
                min_idx = j

        if min_idx != i:

            arr[i], arr[min_idx] = arr[min_idx], arr[i]

            print(
                f"Bước {step}: {arr}"
            )

            step += 1

    print("Kết quả:", arr)


# ==========================
# INSERTION SORT
# ==========================
def insertion_sort(arr):

    print("\n===== INSERTION SORT =====")
    print("Ban đầu:", arr)

    step = 1

    for i in range(1, len(arr)):

        key = arr[i]

        j = i - 1

        while j >= 0 and arr[j] > key:

            arr[j + 1] = arr[j]

            j -= 1

        arr[j + 1] = key

        print(
            f"Bước {step}: {arr}"
        )

        step += 1

    print("Kết quả:", arr)


# ==========================
# MAIN
# ==========================

data = [5, 3, 8, 4, 2]

bubble_sort(data.copy())

selection_sort(data.copy())

insertion_sort(data.copy())

### Bài 3: Best / Worst / Average Case ⭐⭐
Với mảng đã sắp xếp, ngược chiều, và ngẫu nhiên — đo thời gian chạy 3 thuật toán. Kết luận khi nào nên dùng cái nào.
CÂU TRẢ LỜI:
import random
import time

# ==========================
# Bubble Sort
# ==========================
def bubble_sort(arr):
    n = len(arr)

    for i in range(n):
        for j in range(n - i - 1):

            if arr[j] > arr[j + 1]:
                arr[j], arr[j + 1] = arr[j + 1], arr[j]


# ==========================
# Selection Sort
# ==========================
def selection_sort(arr):
    n = len(arr)

    for i in range(n):

        min_idx = i

        for j in range(i + 1, n):

            if arr[j] < arr[min_idx]:
                min_idx = j

        arr[i], arr[min_idx] = arr[min_idx], arr[i]


# ==========================
# Insertion Sort
# ==========================
def insertion_sort(arr):

    for i in range(1, len(arr)):

        key = arr[i]

        j = i - 1

        while j >= 0 and arr[j] > key:

            arr[j + 1] = arr[j]

            j -= 1

        arr[j + 1] = key


# ==========================
# Hàm đo thời gian
# ==========================
def benchmark(sort_func, data):

    arr = data.copy()

    start = time.perf_counter()

    sort_func(arr)

    end = time.perf_counter()

    return end - start


# ==========================
# Sinh dữ liệu
# ==========================
N = 1000

# Best Case
best_case = list(range(N))

# Worst Case
worst_case = list(range(N, 0, -1))

# Average Case
average_case = random.sample(range(N * 10), N)

# ==========================
# Danh sách thuật toán
# ==========================
algorithms = [
    ("Bubble", bubble_sort),
    ("Selection", selection_sort),
    ("Insertion", insertion_sort)
]

# ==========================
# Danh sách bộ test
# ==========================
test_cases = [
    ("Best Case", best_case),
    ("Worst Case", worst_case),
    ("Average Case", average_case)
]

# ==========================
# Chạy benchmark
# ==========================
for case_name, data in test_cases:

    print("\n" + "=" * 60)
    print(case_name.center(60))
    print("=" * 60)

    print(f"{'Thuật toán':<15}{'Thời gian (giây)'}")
    print("-" * 35)

    for name, func in algorithms:

        runtime = benchmark(func, data)

        print(
            f"{name:<15}"
            f"{runtime:.6f}"
        )

CÂU TRẢ LỜI
import os
import time
import random

RED = "\033[91m"
GREEN = "\033[92m"
CYAN = "\033[96m"
YELLOW = "\033[93m"
RESET = "\033[0m"

DELAY = 0.2


def clear():
    os.system(
        "cls" if os.name == "nt"
        else "clear"
    )


def draw_array(arr,
               current1=-1,
               current2=-1,
               step=0,
               total=0,
               title=""):

    clear()

    print(
        f"{CYAN}{title}{RESET}"
    )

    print(
        f"Bước {step}/{total}\n"
    )

    for i, value in enumerate(arr):

        bar = "█" * value

        if i == current1 or i == current2:

            print(
                f"{RED}{bar} {value}{RESET}"
            )

        else:

            print(
                f"{bar} {value}"
            )

    print()

    if current1 != -1:

        print(
            f"{YELLOW}Đang so sánh:"
            f" [{current1}] và [{current2}]"
            f"{RESET}"
        )

    print(
        f"Số bước còn lại: "
        f"{total-step}"
    )


def bubble_visual(arr):

    n = len(arr)

    total = n * (n - 1) // 2

    step = 0

    for i in range(n):

        for j in range(n - i - 1):

            step += 1

            draw_array(
                arr,
                j,
                j + 1,
                step,
                total,
                "Bubble Sort"
            )

            time.sleep(DELAY)

            if arr[j] > arr[j + 1]:

                arr[j], arr[j + 1] = (
                    arr[j + 1],
                    arr[j]
                )

    clear()

    print(
        f"{GREEN}Đã sắp xếp xong!{RESET}"
    )

    print(arr)


def main():

    print("1. Bubble Sort")
    print("2. Selection Sort")
    print("3. Insertion Sort")

    choice = input(
        "\nChọn thuật toán: "
    )

    data = [
        random.randint(3, 25)
        for _ in range(8)
    ]

    if choice == "1":

        bubble_visual(data)

    else:

        print(
            "Selection và Insertion "
            "có thể triển khai tương tự."
        )


main()
